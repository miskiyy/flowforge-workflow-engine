Query (matches execution/repository.ts listRuns, tenant/workflow/status
filter, keyset-paginated newest-first, limit+1 = 21):

  select id, tenant_id, workflow_id, workflow_version_id, trigger_type,
         triggered_by, status, started_at, finished_at, created_at
  from runs
  where tenant_id = '11111111-1111-1111-1111-111111111111'
    and workflow_id = '46b06970-7796-41f1-86d7-1a1d4968e4a8'
    and status = 'failed'
  order by created_at desc, id desc
  limit 21;
